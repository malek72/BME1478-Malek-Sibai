#Looping through every folder that starts with the name "2018" (looping through the number of folders avalaible)
# Than getting inside that folder and than for every file that contains jpg change theri nnames with the name of the folder they are in
for i in 2018*
do
    cd "$i"

    for k in *.jpg
    do
        mv "$k" "$i-$k"
    done

    cd ..
done

# Copies all images with .jpg to directory IMAGES
cp 2018*/*.jpg ../IMAGES

# Copy the eeg the txt and csv files to the folder DATA_ANALYSIS

do

done
